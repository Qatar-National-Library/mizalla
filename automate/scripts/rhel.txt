# os_version
cat /etc/os-release 2>/dev/null

# ram
cat /proc/meminfo 2>/dev/null

# disk
df -k / 2>/dev/null | tail -n 1

# external_storage
findmnt -rn -o TARGET,FSTYPE,SOURCE 2>/dev/null | grep -E '/media/|/mnt/|/run/media/|nfs|cifs|smb' | grep -v 'rootfs' | grep -vE '^\s*(/home|/root|/var/lib|/tmp|/run)' | awk '{print $1}' || true
lsblk -rn -o MOUNTPOINT,RM,TRAN 2>/dev/null | awk '($2=="1" || $3=="usb") && $1!="" && $1!~"^/(home|root|var|tmp|run)" {print $1}'

# docker
docker --version 2>/dev/null
docker info 2>/dev/null
find /opt /etc /home /var/www /root -maxdepth 4 \( -name "docker-compose.yml" -o -name "docker-compose.yaml" -o -name "compose.yaml" \) 2>/dev/null | head -n 3

# kubernetes
kubectl version --client 2>/dev/null
kubectl get pods -A 2>/dev/null || kubectl get pods 2>/dev/null

# python
python3 --version 2>/dev/null || python --version 2>/dev/null

# php
php -v 2>/dev/null | head -n 1

# nodejs
node -v 2>/dev/null

# uvicorn
uvicorn --version 2>/dev/null || python3 -m uvicorn --version 2>/dev/null
pgrep -a -f '[u]vicorn' 2>/dev/null

# apache
httpd -v 2>/dev/null || apache2 -v 2>/dev/null
pgrep -x httpd 2>/dev/null || pgrep -x apache2 2>/dev/null

# nginx
nginx -v 2>&1
pgrep -x nginx 2>/dev/null

# drupal
find /var/www /srv /opt /usr/share/nginx/html -maxdepth 4 -name "Drupal.php" -path "*/core/lib/Drupal.php" 2>/dev/null | head -n 1
